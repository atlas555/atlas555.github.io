<script>
var writers = [];
var currentChars = ['彩'];

// Initialize the first writer
document.addEventListener('DOMContentLoaded', function() {
  // Create the first writer
  var writer1 = HanziWriter.create('grid-target-1', currentChars[0], {
    width: 100,
    height: 100,
    padding: 5,
    showOutline: false,
    strokeAnimationSpeed: 1,
    delayBetweenStrokes: 10,
    radicalColor: '#337ab7'
  });
  
  writers.push(writer1);
  
  // Set up event listeners
  document.getElementById('animate-all-button').addEventListener('click', function() {
    animateAllCharacters();
  });
  
  document.getElementById('submit-button').addEventListener('click', updateCharacters);
  
  document.getElementById('char-input').addEventListener('keypress', function(e) {
    if (e.key === 'Enter') {
      updateCharacters();
    }
  });
});

// Function to animate all characters
function animateAllCharacters() {
  for (var i = 0; i < writers.length; i++) {
    (function(index) {
      setTimeout(function() {
        writers[index].animateCharacter();
      }, index * 1000); // Stagger animations by 1 second
    })(i);
  }
}

// Function to update the character display
function updateCharacters() {
  var charInput = document.getElementById('char-input').value;
  if (charInput && charInput.length > 0) {
    // Take up to 3 characters
    var newChars = charInput.substring(0, 3).split('');
    currentChars = newChars;
    
    // Clear existing writers and displays
    writers = [];
    var charactersRow = document.getElementById('characters-row');
    charactersRow.innerHTML = '';
    
    // Create new displays and writers for each character
    for (var i = 0; i < newChars.length; i++) {
      // Create a new character display
      var charDisplay = document.createElement('div');
      charDisplay.className = 'character-display';
      
      // Create SVG for the grid
      var svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
      svg.setAttribute('xmlns', 'http://www.w3.org/2000/svg');
      svg.setAttribute('width', '100');
      svg.setAttribute('height', '100');
      svg.setAttribute('id', 'grid-target-' + (i + 1));
      
      // Add grid lines
      var lines = [
        {x1: 0, y1: 0, x2: 100, y2: 100},
        {x1: 100, y1: 0, x2: 0, y2: 100},
        {x1: 50, y1: 0, x2: 50, y2: 100},
        {x1: 0, y1: 50, x2: 100, y2: 50}
      ];
      
      for (var j = 0; j < lines.length; j++) {
        var line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
        line.setAttribute('x1', lines[j].x1);
        line.setAttribute('y1', lines[j].y1);
        line.setAttribute('x2', lines[j].x2);
        line.setAttribute('y2', lines[j].y2);
        line.setAttribute('stroke', '#DDD');
        svg.appendChild(line);
      }
      
      charDisplay.appendChild(svg);
      charactersRow.appendChild(charDisplay);
      
      // Create a new writer for this character
      var writer = HanziWriter.create('grid-target-' + (i + 1), newChars[i], {
        width: 100,
        height: 100,
        padding: 5,
        showCharacter: false
        // showOutline: false,
        strokeAnimationSpeed: 1,
        delayBetweenStrokes: 20,
        radicalColor: '#337ab7'
      });
      
      writers.push(writer);
    }
    
    // Clear the input field
    document.getElementById('char-input').value = '';
    
    // Animate all characters with a slight delay between each
    animateAllCharacters();
  }
}
</script>
