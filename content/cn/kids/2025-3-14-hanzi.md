<script>
// Global variables
const writerOptions = {
  width: 100,
  height: 100,
  padding: 5,
  showOutline: true,
  strokeColor: '#337ab7',
  strokeAnimationSpeed: 1,
  delayBetweenStrokes: 20
};

let writers = [];
let currentChars = ['彩'];

// Initialize when DOM is loaded
document.addEventListener('DOMContentLoaded', function() {
  // Create the first writer
  const writer = HanziWriter.create('grid-target-1', currentChars[0], writerOptions);
  writers.push(writer);
  
  // Add pinyin to the first character
  addPinyinToCharacter('pinyin-1', currentChars[0]);
  
  // Update reference link for the first character
  updateReferenceLink(currentChars[0]);
  
  // Set up event listeners
  document.getElementById('animate-button').addEventListener('click', animateAllCharacters);
  document.getElementById('submit-button').addEventListener('click', updateCharacters);
  
  // Allow pressing Enter in the input field
  document.getElementById('char-input').addEventListener('keypress', function(e) {
    if (e.key === 'Enter') {
      updateCharacters();
    }
  });
});

// Get pinyin for a character
function getPinyin(char) {
  if (!char) return '';
  // Use the tiny-pinyin library to get the pinyin
  return window.pinyin.convertToPinyin(char);
}

// Add pinyin text above a character
function addPinyinToCharacter(id, char) {
  const pinyinElement = document.getElementById(id);
  if (pinyinElement) {
    pinyinElement.textContent = getPinyin(char);
  }
}

// Update the reference link based on the current character
function updateReferenceLink(char) {
  const linkContainer = document.getElementById('reference-link');
  if (linkContainer && char) {
    linkContainer.innerHTML = `更多资料：汉典（<a href="https://www.zdic.net/hans/${char}" target="_blank">查看详情</a>）`;
  }
}

// Animate characters sequentially
function animateAllCharacters() {
  if (writers.length === 0) return;
  animateCharacterSequentially(0);
}

// Recursive function to animate characters one after another
function animateCharacterSequentially(index) {
  if (index >= writers.length) return;
  
  writers[index].animateCharacter({
    onComplete: function() {
      setTimeout(() => animateCharacterSequentially(index + 1), 500);
    }
  });
}

// Create grid lines for SVG
function createGridLines() {
  return [
    {x1: 0, y1: 0, x2: 100, y2: 100},
    {x1: 100, y1: 0, x2: 0, y2: 100},
    {x1: 50, y1: 0, x2: 50, y2: 100},
    {x1: 0, y1: 50, x2: 100, y2: 50}
  ];
}

// Create SVG element with grid
function createSvgGrid(id) {
  const svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
  svg.setAttribute('xmlns', 'http://www.w3.org/2000/svg');
  svg.setAttribute('width', '100');
  svg.setAttribute('height', '100');
  svg.setAttribute('id', id);
  
  // Add grid lines
  const lines = createGridLines();
  lines.forEach(line => {
    const lineElement = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    lineElement.setAttribute('x1', line.x1);
    lineElement.setAttribute('y1', line.y1);
    lineElement.setAttribute('x2', line.x2);
    lineElement.setAttribute('y2', line.y2);
    lineElement.setAttribute('stroke', '#DDD');
    svg.appendChild(lineElement);
  });
  
  return svg;
}

// Update characters based on input
function updateCharacters() {
  const charInput = document.getElementById('char-input').value;
  if (!charInput) return;
  
  // Take up to 3 characters
  const newChars = charInput.substring(0, 3).split('');
  currentChars = newChars;
  
  // Clear existing writers and displays
  writers = [];
  const charactersRow = document.getElementById('characters-row');
  charactersRow.innerHTML = '';
  
  // Create new displays and writers for each character
  newChars.forEach((char, i) => {
    // Create display container
    const charDisplay = document.createElement('div');
    charDisplay.className = 'character-display';
    
    // Add pinyin element
    const pinyinElement = document.createElement('div');
    pinyinElement.className = 'pinyin-text';
    pinyinElement.id = `pinyin-${i + 1}`;
    pinyinElement.textContent = getPinyin(char);
    charDisplay.appendChild(pinyinElement);
    
    // Create SVG grid
    const gridId = `grid-target-${i + 1}`;
    const svg = createSvgGrid(gridId);
    charDisplay.appendChild(svg);
    charactersRow.appendChild(charDisplay);
    
    // Create writer for this character
    const writer = HanziWriter.create(gridId, char, writerOptions);
    writers.push(writer);
  });
  
  // Clear the input field
  document.getElementById('char-input').value = '';
  
  // Update reference link with the first character
  if (newChars.length > 0) {
    updateReferenceLink(newChars[0]);
  }
  
  // Animate all characters
  animateAllCharacters();
}
</script>
