<script>
var currentChar = '彩';
var writer = HanziWriter.create('grid-background-target', currentChar, {
  width: 100,
  height: 100,
  padding: 5,
  showOutline: false,
  strokeAnimationSpeed: 1, // 5x normal speed
  delayBetweenStrokes: 10, // milliseconds
  radicalColor: '#337ab7' // blue
});

// Character data dictionary with pinyin and meaning
var characterData = {
  '彩': { pinyin: 'cǎi', meaning: '彩色，色彩' },
  '学': { pinyin: 'xué', meaning: '学习，学校' },
  '字': { pinyin: 'zì', meaning: '文字，字体' },
  '好': { pinyin: 'hǎo', meaning: '好的，良好' },
  '爱': { pinyin: 'ài', meaning: '爱心，喜爱' },
  '家': { pinyin: 'jiā', meaning: '家庭，家人' },
  '国': { pinyin: 'guó', meaning: '国家，国度' },
  '人': { pinyin: 'rén', meaning: '人类，人物' },
  '天': { pinyin: 'tiān', meaning: '天空，天气' },
  '地': { pinyin: 'dì', meaning: '地面，土地' }
};

document.getElementById('animate-button').addEventListener('click', function() {
    writer.animateCharacter();
});

// Function to update the character display
function updateCharacter() {
  var charInput = document.getElementById('char-input').value;
  if (charInput && charInput.length > 0) {
    // Take only the first character if multiple are entered
    var newChar = charInput.charAt(0);
    currentChar = newChar;
    
    // Update the writer with the new character
    writer.setCharacter(newChar);
    
    // Update the character info
    document.getElementById('display-char').textContent = newChar;
    
    // Update pinyin and meaning if available in our dictionary
    if (characterData[newChar]) {
      document.getElementById('display-pinyin').textContent = characterData[newChar].pinyin;
      document.getElementById('display-meaning').textContent = characterData[newChar].meaning;
      document.getElementById('character-meaning').style.display = 'block';
    } else {
      // If character not in our dictionary, show a default message
      document.getElementById('display-pinyin').textContent = '未知';
      document.getElementById('display-meaning').textContent = '未收录';
      document.getElementById('character-meaning').style.display = 'block';
    }
    
    // Clear the input field
    document.getElementById('char-input').value = '';
    
    // Animate the new character
    writer.animateCharacter();
  }
}

// Add event listener after DOM is fully loaded
document.addEventListener('DOMContentLoaded', function() {
  document.getElementById('submit-button').addEventListener('click', updateCharacter);
  
  // Allow pressing Enter in the input field
  document.getElementById('char-input').addEventListener('keypress', function(e) {
    if (e.key === 'Enter') {
      updateCharacter();
    }
  });
  
  // Initialize with the default character
  if (characterData[currentChar]) {
    document.getElementById('display-pinyin').textContent = characterData[currentChar].pinyin;
    document.getElementById('display-meaning').textContent = characterData[currentChar].meaning;
  }
});
</script>
